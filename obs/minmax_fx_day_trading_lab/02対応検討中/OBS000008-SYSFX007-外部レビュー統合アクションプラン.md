### 4.4 叩き台 3-A: `aggregate_to_json` 改修ドラフト

```python
# scripts/data/fetch_ds1_ohlcv.py:127-155 を以下に置換

def aggregate_to_json(raw_dir: Path, out_json: Path) -> dict:
    """CSV 群を 1 つの JSON に集約 (同一通貨の複数ファイルを結合)."""
    files = sorted(raw_dir.glob("ohlcv_*.csv"))
    # symbol ごとに複数ファイルを蓄積
    pair_frames: dict[str, list[pd.DataFrame]] = {}
    for f in files:
        df = pd.read_csv(f, index_col="timestamp", parse_dates=True)
        name = f.stem
        parts = name.split("_")
        # ohlcv_USD_JPY_5min_20200101_... → parts = [ohlcv, USD, JPY, 5min, ...]
        symbol = "_".join(parts[1:3])
        pair_frames.setdefault(symbol, []).append(df)

    out: dict = {
        "schema_version": "1.0",
        "generated_at": datetime.now().isoformat(),
        "interval": INTERVAL,
        "pairs": {},
    }
    for symbol, dfs in pair_frames.items():
        # 同一通貨の複数ファイルを concat → 重複削除 → ソート
        combined = pd.concat(dfs).sort_index()
        combined = combined[~combined.index.duplicated(keep="last")]
        out["pairs"][symbol] = {
            "n_bars": len(combined),
            "start": combined.index[0].isoformat(),
            "end": combined.index[-1].isoformat(),
            "columns": list(combined.columns),
            "data": combined.reset_index().to_dict(orient="records"),
        }
    out_json.parent.mkdir(parents=True, exist_ok=True)
    out_json.write_text(
        json.dumps(out, indent=2, ensure_ascii=False, default=str),
        encoding="utf-8",
    )
    return {
        "n_pairs": len(pair_frames),
        "total_bars": sum(len(d) for d in pair_frames.values()),
        "json": str(out_json),
    }
```

### 7.1 目的

### 7.2 新規スクリプト骨子: `scripts/analyze_market_characteristics.py`

```python
"""通貨ペア別市場特性分析 (Phase 1 叩き台, OBS000006 Phase 1).

5 通貨 × Train データ (2020-2022) のみを対象に以下を算出:
  1. H4 ATR(14) の分布 (中央値, 25/50/75/90 パーセンタイル, pips 換算)
  2. M5 バーの高安差 (ノイズ床) の分布
  3. Donchian チャネル内での価格滞留期間 (ラン長) の分布
  4. ADX(14) のヒストグラム
  5. ブレイク後フォロースルー率 (ブレイク後に X×ATR 以上動いた割合)

出力: data/curated/pair_characteristics.json
"""
from __future__ import annotations
import argparse
import json
import sys
from pathlib import Path
import numpy as np
import pandas as pd

ROOT = Path(__file__).resolve().parents[1]
SRC = ROOT / "src"
sys.path.insert(0, str(SRC))

from minmax_fx_dt.data import GMOClient
from minmax_fx_dt.strategy.indicators import atr, adx, donchian

PAIRS = ["USD_JPY", "EUR_JPY", "GBP_JPY", "AUD_JPY", "EUR_USD"]
TRAIN_START = "2020-01-01"
TRAIN_END = "2022-12-31"


def pip_value(is_jpy: bool) -> float:
    return 0.01 if is_jpy else 0.0001


def aggregate_to_h4(m5: pd.DataFrame) -> pd.DataFrame:
    return pd.DataFrame({
        "open": m5["open"].resample("4h").first(),
        "high": m5["high"].resample("4h").max(),
        "low": m5["low"].resample("4h").min(),
        "close": m5["close"].resample("4h").last(),
    }).dropna()


def aggregate_to_d1(m5: pd.DataFrame) -> pd.DataFrame:
    return pd.DataFrame({
        "open": m5["open"].resample("D").first(),
        "high": m5["high"].resample("D").max(),
        "low": m5["low"].resample("D").min(),
        "close": m5["close"].resample("D").last(),
    }).dropna()


def analyze_pair(symbol: str, m5: pd.DataFrame) -> dict:
    is_jpy = "JPY" in symbol
    pv = pip_value(is_jpy)
    h4 = aggregate_to_h4(m5)
    d1 = aggregate_to_d1(m5)

    # 1. H4 ATR(14) 分布
    h4_atr = atr(h4["high"], h4["low"], h4["close"], length=14).dropna()
    h4_atr_pips = h4_atr / pv
    h4_atr_stats = {
        "median": float(h4_atr_pips.median()),
        "p25": float(h4_atr_pips.quantile(0.25)),
        "p50": float(h4_atr_pips.quantile(0.50)),
        "p75": float(h4_atr_pips.quantile(0.75)),
        "p90": float(h4_atr_pips.quantile(0.90)),
    }

    # 2. M5 ノイズ床 (高安差) 分布
    m5_range = (m5["high"] - m5["low"]) / pv
    noise_stats = {
        "median": float(m5_range.median()),
        "p95": float(m5_range.quantile(0.95)),
    }

    # 3. Donchian(20) チャネル内ラン長
    dc = donchian(h4["high"], h4["low"], lower_length=20, upper_length=20)
    in_channel = (h4["close"] >= dc["DCL"]) & (h4["close"] <= dc["DCU"])
    # 連続 True の長さ
    run_lengths = []
    cur = 0
    for v in in_channel:
        if v:
            cur += 1
        else:
            if cur > 0:
                run_lengths.append(cur)
            cur = 0
    if cur > 0:
        run_lengths.append(cur)
    run_stats = {
        "median": float(np.median(run_lengths)) if run_lengths else 0.0,
        "p75": float(np.percentile(run_lengths, 75)) if run_lengths else 0.0,
    }

    # 4. D1 ADX(14) ヒストグラム
    adx_df = adx(d1["high"], d1["low"], d1["close"], length=14)
    adx_val = adx_df.iloc[:, 0].dropna()
    adx_stats = {
        "median": float(adx_val.median()),
        "p70": float(adx_val.quantile(0.70)),
        "p90": float(adx_val.quantile(0.90)),
    }

    # 5. ブレイク後フォロースルー率 (H4 Donchian(20) 上限ブレイク後 5 本以内に 1×ATR 以上動いた割合)
    upper_break = h4["close"] > dc["DCU"].shift(1)
    follow_through_pct = 0.0
    if upper_break.sum() > 0:
        follow_count = 0
        for i in upper_break[upper_break].index:
            idx = h4.index.get_loc(i)
            if idx + 5 < len(h4):
                window = h4.iloc[idx + 1: idx + 6]
                move = (window["high"].max() - h4.loc[i, "close"]) / h4_atr.loc[i]
                if move >= 1.0:
                    follow_count += 1
        follow_through_pct = follow_count / upper_break.sum()

    return {
        "symbol": symbol,
        "is_jpy": is_jpy,
        "pip_value": pv,
        "h4_atr_pips": h4_atr_stats,
        "m5_noise_pips": noise_stats,
        "donchian20_run_length_bars": run_stats,
        "d1_adx14": adx_stats,
        "h4_breakout_followthrough_pct": float(follow_through_pct),
    }


def main() -> int:
    parser = argparse.ArgumentParser(description="通貨ペア別市場特性分析 (Train 2020-2022)")
    parser.add_argument("--out", default="data/curated/pair_characteristics.json")
    args = parser.parse_args()

    # TODO: Phase -1 完了後、ds-1.json から Train 期間のみ抽出
    # 当面は各通貨の CSV から直接読み込み
    results = {}
    for symbol in PAIRS:
        csv_path = ROOT / "data" / "raw" / "ds-1" / f"ohlcv_{symbol}_5min_20200101_20240630_dukascopy.csv"
        if not csv_path.exists():
            print(f"  [NG] {symbol}: CSV なし ({csv_path})")
            continue
        m5 = pd.read_csv(csv_path, index_col="timestamp", parse_dates=True)
        m5 = m5[(m5.index >= TRAIN_START) & (m5.index <= TRAIN_END)]
        print(f"[{symbol}] Train bars: {len(m5)} ({m5.index[0].date()} - {m5.index[-1].date()})")
        results[symbol] = analyze_pair(symbol, m5)

    out_path = ROOT / args.out
    out_path.parent.mkdir(parents=True, exist_ok=True)
    out_path.write_text(json.dumps(results, indent=2, ensure_ascii=False), encoding="utf-8")
    print(f"\nWROTE: {out_path} ({len(results)} 通貨)")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

### 8.3 permutation test 実装

```python
# src/minmax_fx_dt/decision/permutation.py 骨子

"""Permutation test for strategy significance.

5 通貨 × 複数プリセットの探索で「最も良かった結果」が
偶然かどうかを 1000 回のシャッフルで検定する。

使い方:
  from minmax_fx_dt.decision.permutation import permutation_test_pvalue
  p = permutation_test_pvalue(monthly_returns, n_permutations=1000, random_state=42)
"""
from __future__ import annotations
import numpy as np


def permutation_test_pvalue(
    observed_metric: float,
    monthly_returns: np.ndarray,
    n_permutations: int = 1000,
    random_state: int = 42,
) -> float:
    """Permutation p 値.

    Args:
        observed_metric: 観測された月次 Sharpe (例: 0.5).
        monthly_returns: 元の月次リターン配列.
        n_permutations: シャッフル回数.
        random_state: 再現性シード.

    Returns:
        p 値 (観測値以上のメトリックが偶然得られる確率).
    """
    rng = np.random.default_rng(random_state)
    count_ge = 0
    for _ in range(n_permutations):
        shuffled = rng.permutation(monthly_returns)
        if shuffled.std() > 0:
            perm_sharpe = shuffled.mean() / shuffled.std() * np.sqrt(12)
        else:
            perm_sharpe = 0.0
        if perm_sharpe >= observed_metric:
            count_ge += 1
    return count_ge / n_permutations
```
